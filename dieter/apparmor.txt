#include <tunables/global>

# AppArmor profile of the Dieter add-on. The Supervisor loads it under the add-on's slug
# (the name below is replaced), so no rule may reference the profile by name.
#
# The container runs s6/bashio, the bridge (python3), the Dieter daemon, tmux, git and the
# agent CLIs (claude, codex, agy), and the agents themselves run whatever a task needs
# (shells, node, npm, pip, …) under /data. File access is therefore unrestricted, like
# Docker's default profile; what the profile pins down is the capability set (Docker's
# default set, nothing more), no mounts, and no writes to the kernel interfaces.
profile dieter flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # run.sh: chown -R /data, setpriv to the dieter user; the rest is Docker's default set
  capability audit_write,
  capability chown,
  capability dac_override,
  capability dac_read_search,
  capability fowner,
  capability fsetid,
  capability kill,
  capability mknod,
  capability net_bind_service,
  capability net_raw,
  capability setfcap,
  capability setgid,
  capability setpcap,
  capability setuid,
  capability sys_chroot,

  network,
  file,
  umount,

  # the bridge supervises the daemon, tmux and the agents' processes
  signal (send,receive),
  # the Supervisor/Docker stop the container
  signal (receive) peer=unconfined,
  signal (receive) peer=runc,
  # ps inside the container, node inspector
  ptrace (trace,read,tracedby,readby),

  deny mount,

  # /proc and /sys as Docker's default profile: read everything, write only what a process owns
  deny @{PROC}/* w,
  deny @{PROC}/{[^1-9],[^1-9][^0-9],[^1-9s][^0-9y][^0-9s],[^1-9][^0-9][^0-9][^0-9/]*}/** w,
  deny @{PROC}/sys/[^k]** w,
  deny @{PROC}/sys/kernel/{?,??,[^s][^h][^m]**} w,
  deny @{PROC}/sysrq-trigger rwklx,
  deny @{PROC}/kcore rwklx,
  deny /sys/[^f]*/** wklx,
  deny /sys/f[^s]*/** wklx,
  deny /sys/fs/[^c]*/** wklx,
  deny /sys/fs/c[^g]*/** wklx,
  deny /sys/fs/cg[^r]*/** wklx,
  deny /sys/firmware/** rwklx,
  deny /sys/devices/virtual/powercap/** rwklx,
  deny /sys/kernel/security/** rwklx,
}
